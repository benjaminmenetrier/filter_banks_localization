\section{EnVar formulation}

\subsection{Standard EnVar formulation}
For an ensemble $N$ backgrounds $\left\{\mathbf{x}^1,\dots,\mathbf{x}^N\right\}$, with $\mathbf{x}^k \in \mathbb{R}^n$, the standard EnVar formulation is given by:
\begin{align}
\label{eq:envar}
\mathbf{B} = \mathbf{X}\mathbf{X}^\mathrm{T} \circ \mathbf{L}
\end{align}
where:
\begin{itemize}
\item $\mathbf{B} \in \mathbb{R}^{n \times n}$ is the resulting background error covariance.
\item $\mathbf{X} \in \mathbb{R}^{n \times N}$ contains the normalized ensemble perturbations:
\begin{align}
X_{ij} = \frac{1}{\sqrt{N-1}} \left(x^j_i - \langle x_i \rangle\right)
\end{align}
with $\langle x_i \rangle$ denoting the ensemble mean:
\begin{align}
\langle x_i \rangle = \frac{1}{N} \sum_{j=1}^N x^j_i
\end{align}
\item $\mathbf{L} \in \mathbb{R}^{n \times n}$ is the localization matrix.
\item $\circ$ denotes a Schur product between matrices (element-by-element).
\end{itemize}
The square-root expression of \eqref{eq:envar}, transforming the control vector $\mathbf{v} \in \mathbb{R}^m$ into a control increment $\delta \mathbf{x} \in \mathbb{R}^n$, is given by:
\begin{align}
\delta \mathbf{x} = \sum_{j=1}^N \mathbf{X}_{:j} \circ \left(\mathbf{U} \mathbf{v}_j\right)
\end{align}
where:
\begin{itemize}
\item $\mathbf{X}_{:j} \in \mathbb{R}^n$ is the j$^\text{th}$ column of $\mathbf{X}$
\item $\mathbf{U} \in \mathbb{R}^{n \times m}$ is a square-root of the localization $\mathbf{L}$ such that $\mathbf{U} \mathbf{U}^\mathrm{T} = \mathbf{L}$.
\item $\mathbf{v}_j \in \mathbb{R}^{m/N}$ is the j$^\text{th}$ subvector of $\mathbf{v}$.
\end{itemize}

\subsection{Generalized EnVar formulation}
The standard EnVar formulation can be generalized by introducing two linear transforms $\mathbf{F} \in \mathbb{R}^{n \times \widehat{n}}$ and $\mathbf{F}^\dag \in \mathbb{R}^{\widehat{n} \times n}$:
\begin{align}
\widehat{\mathbf{B}} & = \mathbf{F} \left(\left(\mathbf{F}^\dag \mathbf{X}\right) \left(\mathbf{F}^\dag \mathbf{X}\right)^\mathrm{T} \circ \widehat{\mathbf{L}}\right) \mathbf{F}^\mathrm{T} \\
\delta \mathbf{x} & = \mathbf{F} \sum_{j=1}^N \left(\mathbf{F}^\dag \mathbf{X}_{:j}\right) \circ \left(\widehat{\mathbf{U}} \widehat{\mathbf{v}}_j\right)
\end{align}
where: 
\begin{itemize}
\item $\widehat{\mathbf{L}} \in \mathbb{R}^{\widehat{n} \times \widehat{n}}$ is the localization matrix in the transformed space.
\item $\widehat{\mathbf{U}} \in \mathbb{R}^{n \times \widehat{m}}$ is a square-root of the localization $\widehat{\mathbf{L}}$ such that $\widehat{\mathbf{U}} \widehat{\mathbf{U}}^\mathrm{T} = \widehat{\mathbf{L}}$.
\item $\widehat{\mathbf{v}}_j \in \mathbb{R}^{\widehat{m}/N}$ is the j$^\text{th}$ subvector of $\widehat{\mathbf{v}} \in \mathbb{R}^{\widehat{m}}$.
\end{itemize}
Let's analyse the asymptotic values if the ensemble size grows to infinity:
\begin{itemize}
\item The localization matrices $\mathbf{L}$ and $\widehat{\mathbf{L}}$ should converge to $\mathbf{1}_n$ and $\mathbf{1}_{\widehat{n}}$, respectively, where $\mathbf{1}_n \in \mathbb{R}^{n \times n}$ is a matrix filled with 1.
\item As a consequence:
\begin{align}
\lim_{N \rightarrow \infty} \mathbf{B} & = \mathbf{X} \mathbf{X}^\mathrm{T} \\
\lim_{N \rightarrow \infty} \widehat{\mathbf{B}} & = \left(\mathbf{F} \mathbf{F}^\dag \mathbf{X}\right) \left(\mathbf{F} \mathbf{F}^\dag \mathbf{X}\right)^\mathrm{T}
\end{align}
\item To ensure consistency, linear transforms $\mathbf{F}$ and $\mathbf{F}^\dag$ should not have an impact on the asymptotic background error covariance:
\begin{align}
\lim_{N \rightarrow \infty} \mathbf{B} = \lim_{N \rightarrow \infty} \widehat{\mathbf{B}}
\end{align}
so
\begin{align}
\mathbf{F} \mathbf{F}^\dag \mathbf{X} = \mathbf{X}
\end{align}
Thus, $\mathbf{F}^\dag$ should be a right-inverse of $\mathbf{F}$ in the ensemble perturbations space.
\end{itemize}

The content of $\mathbf{F}$ and $\mathbf{F}^\dag$ can be very diverse:
\begin{itemize}
\item A change of variables: for instance towards a better-suited humidity variable.
\item A inverse balance operator as in \citet{clayton_2012}, where the localization is applied on unbalanced variables and the balance operator is finally applied to get the increment.
\item A change of resolution as in \citet{kleist_2014a}, where the ensemble resolution is lower than the final increment one. In this case, $\mathbf{F}$ is an interpolation from low to high resolution, and the low-resolution perturbations $\mathbf{X}_\mathrm{LR}$ can be thought as the application of a coarsening operator $\mathbf{F}^\dag$ on virtual high-resolution perturbations $\mathbf{X}_\mathrm{HR}$ representing the same field:
\begin{align}
\mathbf{F} \mathbf{X}_\mathrm{LR} = \mathbf{F} \mathbf{F}^\dag \mathbf{X}_\mathrm{HR} = \mathbf{X}_\mathrm{HR}
\end{align}
\end{itemize}

\section{General filter banks}

\subsection{The band-pass framework}
We define a series of $P$ band-pass filters $\left\{\mathbf{H}_1,\dots,\mathbf{H}_P\right\}$, where $\mathbf{H}_p \in \mathbb{R}^{n \times n}$. The linear transform $\mathbf{F}^\dag$ is defined as:
\begin{align}
\mathbf{F}^\dag = \left( \begin{array}{c}
\mathbf{H}_1 \\
\mathbf{H}_2 \\
\vdots \\
\mathbf{H}_P
\end{array}
\right)
\end{align}
and its left-inverse $\mathbf{F}$ as:
\begin{align}
\mathbf{F} = \left(\mathbf{I}_n \mathbf{I}_n \dots \mathbf{I}_n\right)
\end{align}
Thus, $\mathbf{F} \mathbf{F}^\dag$ is the sum of the band-pass filters:
\begin{align}
\mathbf{F} \mathbf{F}^\dag = \sum_{p=1}^P \mathbf{H}_p
\end{align}
and the size of the decomposition space (i.e. the number of rows of $\mathbf{F}^\dag$) is $\widehat{n} = Pn$. For $1 \le p \le P-1$, the band-pass filters $\mathbf{H}_p$ are defined independently. However, to ensure that $\mathbf{F} \mathbf{F}^\dag = \mathbf{I}_n$, the last band-pass filter $\mathbf{H}_P$ is defined as the complement of the sum of all the previous band-pass filters:
\begin{align}
\label{eq:bp1}
\mathbf{H}_P = \mathbf{I}_n - \sum_{p=1}^{P-1} \mathbf{H}_p
\end{align}
We also define the filtered components $\widehat{\mathbf{x}}_p \in \mathbb{R}^n$ as:
\begin{align}
\label{eq:bp2}
\widehat{\mathbf{x}}_p = \mathbf{H}_p \mathbf{x}
\end{align}
where $\mathbf{x} \in \mathbb{R}^n$ is the input vector.

\subsection{Recursive construction}
In this section, we build the band-pass filters $\mathbf{H}_p$ recursively from a series of $P-1$ filters $\left\{\mathbf{G}_1,\dots,\mathbf{G}_{P-1}\right\}$, where $\mathbf{G}_p \in \mathbb{R}^{n \times n}$:
\begin{itemize}
\item The initial residual is given by the input state: $\overline{\mathbf{x}}_0 = \mathbf{x}$
\item At each iteration $1 \le p \le P-1$:
\begin{itemize}
\item The filter $\mathbf{G}_p$ is applied to the previous residual $\overline{\mathbf{x}}_{p-1} \in \mathbb{R}^n$ to produce the new filtered component $\widehat{\mathbf{x}}_p \in \mathbb{R}^{n}$:
\begin{align}
\label{eq:lp1}
\widehat{\mathbf{x}}_p = \mathbf{G}_p \overline{\mathbf{x}}_{p-1}
\end{align}
\item The new residual is computed from $\widehat{\mathbf{x}}_p$:
\begin{align}
\label{eq:lp2}
\overline{\mathbf{x}}_p & = \overline{\mathbf{x}}_{p-1} - \widehat{\mathbf{x}}_p \nonumber \\
& = \left(\mathbf{I}_n - \mathbf{G}_p\right) \overline{\mathbf{x}}_{p-1}
\end{align}
\end{itemize}
\end{itemize}
This recursive construction is illustrated in Figure \ref{fig:algo_recursive}. Integrating recursively, the residual at iteration $p$ becomes:
\begin{align}
\label{eq:lp3}
\overline{\mathbf{x}}_p & = \left(\mathbf{I}_n - \mathbf{G}_p\right) \left(\mathbf{I}_n - \mathbf{G}_{p-1}\right) \dots \left(\mathbf{I}_n - \mathbf{G}_1\right) \mathbf{x} \nonumber \\
& = \displaystyle \prod_{q=1}^p \left(\mathbf{I}_n - \mathbf{G}_q\right) \mathbf{x}
\end{align}
and from equation \eqref{eq:lp1}, we obtain the filtered component at iteration $p$:
\begin{align}
\label{eq:lp4}
\widehat{\mathbf{x}}_p & = \mathbf{G}_p \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right) \mathbf{x}
\end{align}

Combining equations \eqref{eq:bp2} with \eqref{eq:lp4}, we get an equivalence between $\mathbf{H}_p$ and $\mathbf{G}_p$ for $1 \le p \le P-1$:
\begin{align}
\label{eq:band_pass_filter}
\Aboxed{\mathbf{H}_p = \mathbf{G}_p \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right)}
\end{align}
The last filtered component $\widehat{\mathbf{x}}_P$ is actually given by the last residual $\overline{\mathbf{x}}_{P-1}$
\begin{align}
\widehat{\mathbf{x}}_P = \mathbf{H}_P \mathbf{x} = \overline{\mathbf{x}}_{P-1}
\end{align}
So equation \eqref{eq:lp3} gives another expression of equation \eqref{eq:bp1}:
\begin{align}
\Aboxed{\mathbf{H}_P = \prod_{q=1}^{P-1} \left(\mathbf{I}_n - \mathbf{G}_q\right)}
\end{align}

\subsection{Inverse formula}

To inverse equation \eqref{eq:band_pass_filter}, a preliminary step is necessary. We start by recursively showing that for $p > 1$:
\begin{align}
\label{eq:lemna}
\mathbf{I}_n - \sum_{q=1}^{p-1} \mathbf{H}_q = \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right)
\end{align}

The initial case $p = 2$ is obvious: since $\mathbf{H}_1 = \mathbf{G}_1$ (equation \eqref{eq:band_pass_filter}), then $\mathbf{I}_n - \mathbf{H}_1 = \mathbf{I}_n - \mathbf{G}_1$. We now assume that equation \eqref{eq:lemna} holds for a given $p > 1$. As a consequence:
\begin{align}
\mathbf{I}_n - \sum_{q=1}^p \mathbf{H}_q & = \mathbf{I}_n - \sum_{q=1}^{p-1} \mathbf{H}_q - \mathbf{H}_p \nonumber \\
& = \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right) - \mathbf{G}_p \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right) \nonumber \\
& = \left(\mathbf{I}_n - \mathbf{G}_p\right) \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right) \nonumber \\
& = \prod_{q=1}^p \left(\mathbf{I}_n - \mathbf{G}_q\right)
\end{align}
so equation \eqref{eq:lemna} also holds for $p+1$. We can conclude that \eqref{eq:lemna} holds for any $p > 2$. Inserting it into equation \eqref{eq:band_pass_filter}, we get:
\begin{align}
\mathbf{H}_p = \mathbf{G}_p \left(\mathbf{I}_n - \sum_{q=1}^{p-1} \mathbf{H}_q\right)
\end{align}
If $\displaystyle \left(\mathbf{I}_n - \sum_{q=1}^{p-1} \mathbf{H}_q\right)$ has a right inverse, then the filter $\mathbf{G}_p$ can be deduced from the band-pass filters:
\begin{align}
\Aboxed{\mathbf{G}_p = \mathbf{H}_p \left(\mathbf{I}_n - \sum_{q=1}^{p-1} \mathbf{H}_q\right)^{-1}}
\end{align}

\subsection{Embedded filters}
The special case of embedded filters arises when for any $q > p$:
\begin{align}
\label{eq:embedded}
\mathbf{G}_p \mathbf{G}_q = \mathbf{G}_q \mathbf{G}_p = \mathbf{G}_q
\end{align}
The corresponding band-pass filters can then be expressed for $1 \le p \le P-1$ as:
\begin{align}
\mathbf{H}_p & = \mathbf{G}_p \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}_q\right) \nonumber \\
& = \left(\mathbf{G}_p - \mathbf{G}_p \mathbf{G}_{p-1}\right) \prod_{q=1}^{p-2} \left(\mathbf{I}_n - \mathbf{G}_q\right) \nonumber \\
& = \left(\mathbf{G}_p - \mathbf{G}_{p-1}\right) \prod_{q=1}^{p-2} \left(\mathbf{I}_n - \mathbf{G}_q\right)
\end{align}
and since for each factor of $\displaystyle \prod_{q=1}^{p-2} \left(\mathbf{I}_n - \mathbf{G}_q\right)$:
\begin{align}
\left(\mathbf{G}_p - \mathbf{G}_{p-1}\right) \left(\mathbf{I}_n - \mathbf{G}_q\right) & = \mathbf{G}_p - \mathbf{G}_{p-1} - \left(\mathbf{G}_p \mathbf{G}_q - \mathbf{G}_{p-1} \mathbf{G}_q \right) \nonumber \\
& = \mathbf{G}_p - \mathbf{G}_{p-1} - \left(\mathbf{G}_q - \mathbf{G}_q \right) \nonumber \\
& = \mathbf{G}_p - \mathbf{G}_{p-1}
\end{align}
we finally get:
\begin{align}
\Aboxed{\mathbf{H}_p = \mathbf{G}_p - \mathbf{G}_{p-1}}
\end{align}

As a consequence, the construction of a sequence of embedded filters can done in parallel, as illustrated in Figure \ref{fig:algo_parallel}.

\section{Real-case filters}

\subsection{Low-pass and high-pass filters}

For a sequence of filters, we assume that there is an invertible transform $\mathbf{S} \in \mathbb{R}^{n \times n}$ such that every component can be expressed as:
\begin{align}
\mathbf{G}_p = \mathbf{S}^{-1} \mathbf{D}_p \mathbf{S}
\end{align}
where $\mathbf{D}_p \in \mathbb{R}^{n \times n}$ is a diagonal matrix with diagonal elements between 0 and 1. The spectrum $\mathbf{d}_p \in \mathbb{R}^n$ contains all these diagonal elements: $D_{p,ii} = d_{p,i}$.\\
$  $\\
For a sequence of low-pass filters, all the spectra start from 1 and are monotonically decreasing. Symmetrically for a sequence of high-pass filters, all the spectra are monotonically increasing and end up at 1. In both cases, for any $p > 1$ and any element: $d_{p,i} \ge d_{p-1,i}$, as illustrated in Figure \ref{fig:filters_equivalence_low-high}.

\subsection{High-pass filters built from low-pass filters}
Often in practice, only low-pass filters $\mathbf{G}^l_p$ are actually implemented, through a diffusion process or an equivalent smoother. However, the high-pass filter $\mathbf{G}^h_p$ can be built as the complement of the low-pass filter sorted in opposite order $\mathbf{G}^l_{P-p}$:
\begin{align}
\mathbf{G}^h_p = \mathbf{I}_n - \mathbf{G}^l_{P-p}
\end{align}

Updating equation \eqref{eq:band_pass_filter}, the band-pass filter $\mathbf{H}^h_p$ corresponding to the high-pass filters is given for $1 \le p \le P-1$ by:
\begin{align}
\label{eq:band_pass_filter_hp}
\mathbf{H}^h_p & = \mathbf{G}^h_p \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{G}^h_q\right) \nonumber \\
& = \left(\mathbf{I}_n - \mathbf{G}^l_{P-p}\right) \prod_{q=1}^{p-1} \mathbf{G}^l_{P-q}
\end{align}

Figure \ref{fig:filters_equivalence} shows explicit low-pass filters and their complementary high-pass filters counterparts. The corresponding band-pass filters are also displayed in both cases. We can see that the band-pass filters have roughly the same structure in both cases, but are not identical.

\subsection{Embedded filters}
If a sequence of low-pass or high-pass filters is embedded, equation \eqref{eq:embedded} can be simplified for $q < p$ into:
\begin{align}
d_{p,i} d_{q,i} = d_{q,i}
\end{align}
which is only true if:
\begin{align}
\left\{ \begin{array}{rl}
\text{either } & d_{q,i} = 0 \\
\text{or } & d_{p,i} = 1
\end{array}
\right.
\end{align}
A sequence of embedded low-pass filters is illustrated on Figure \ref{fig:filters_equivalence_embedded}. If the sequence of low-pass filters is embedded, it is easy to see that the sequence of complementary high-pass filters is embedded too, and the corresponding band-pass filters can be derived from equation \eqref{eq:band_pass_filter_hp}:
\begin{align}
\mathbf{H}^h_p & = \left(\mathbf{I}_n - \mathbf{G}^l_{P-p}\right) \prod_{q=1}^{p-1} \mathbf{G}^l_{P-q} \nonumber \\
& = \mathbf{G}^l_{P-p+1} - \mathbf{G}^l_{P-p}
\end{align}
So the corresponding band-pass filters are equivalent for the low-pass filters and their high-pass complementary filters (sorted in opposite order):
\begin{align}
\Aboxed{\mathbf{H}^h_p = \mathbf{H}^l_{P-p+1}}
\end{align}

It seems that the difference between a sequence of low-pass filters and the sequence of the complementary high-pass filters becomes greater when more than two bands of the corresponding band-pass filters overlap. In the case of embedded filters where only two bands overlap, this difference vanishes.

\subsection{Pseudo-embedded filters}
Embedded filter are appealing because they can be applied in parallel (see Figure \ref{fig:algo_parallel}). However, low-pass filters based on a diffusion process or an equivalent smoother are not able to produce this kind of filters. The corresponding convolution functions in grid-point space induced by these methods are generally positive, whereas embedded filters require negative lobes, as illustrated in Figure \ref{fig:band_pass_functions}.\\
$  $\\
A possible option is to use the parallel scheme of Figure \ref{fig:algo_parallel} with non-embedded filters. If the different components of the sequence of low-pass filters are sufficiently separated, the difference with the recursive scheme of Figure \ref{fig:algo_recursive} is minor, as illustrated in Figure \ref{fig:filters_equivalence_pseudo-embedded}.

\subsection{Reduced resolution filters}

For a sequence of low-pass filters, it is possible to reduce the memory and computational cost by storing their output at a lower resolution. For this, we redefine the band-pass filter output space: $\mathbf{H}_p \in \mathbb{R}^{\widehat{n}_p \times n}$ where:
\begin{itemize}
\item $\widehat{n}_q \le \widehat{n}_p$ for $q < p$,
\item $\widehat{n}_P = n$ for the last component.
\end{itemize}
The linear transform $\mathbf{F}^\dag$ is still defined as:
\begin{align}
\mathbf{F}^\dag = \left( \begin{array}{c}
\mathbf{H}_1 \\
\mathbf{H}_2 \\
\vdots \\
\mathbf{H}_P
\end{array}
\right)
\end{align}
but its left-inverse is redefined as:
\begin{align}
\mathbf{F} = \left(\mathbf{S}_{P \leftarrow 1} \mathbf{S}_{P \leftarrow 2} \dots \mathbf{I}_n\right)
\end{align}
where $\mathbf{S}_{p \leftarrow q} \in \mathbb{R}^{\widehat{n}_p \times \widehat{n}_q}$ is an interpolator from the resolution $\widehat{n}_q$ to the resolution $\widehat{n}_p$. Thus, $\mathbf{F} \mathbf{F}^\dag$ is the sum of the band-pass filters interpolated at full resolution:
\begin{align}
\mathbf{F} \mathbf{F}^\dag = \sum_{p=1}^P \mathbf{S}_{P \leftarrow p} \mathbf{H}_p
\end{align}
and the size of the decomposition space (i.e. the number of rows of $\mathbf{F}^\dag$) is the sum of all the sizes of the components:
\begin{align}
\widehat{n} = \sum_{p=1}^P \widehat{n}_p
\end{align}

From the filters $\mathbf{G}_p \in \mathbb{R}^{\widehat{n}_p \times n}$, the band-pass filters $\mathbf{H}_p$ are computed for $1 \le p \le P-1$ as:
\begin{align}
\label{eq:band_pass_filter_lr}
\Aboxed{\mathbf{H}_p = \mathbf{G}_p \prod_{q=1}^{p-1} \left(\mathbf{I}_n - \mathbf{S}_{P \leftarrow q} \mathbf{G}_q\right)}
\end{align}
and the last component as:
\begin{align}
\Aboxed{\mathbf{H}_P = \prod_{q=1}^{P-1} \left(\mathbf{I}_n - \mathbf{S}_{P \leftarrow q} \mathbf{G}_q\right)}
\end{align}
The updated recursive scheme is given by Figure \ref{fig:algo_recursive_lr}.\\
$  $\\
For the embedded filters, the assumption \eqref{eq:embedded} should include the interpolators:
\begin{align}
\mathbf{G}_p \mathbf{S}_{P \leftarrow q} \mathbf{G}_q = \mathbf{S}_{p \leftarrow q} \mathbf{G}_q \mathbf{S}_{P \leftarrow p} \mathbf{G}_p = \mathbf{S}_{p \leftarrow q} \mathbf{G}_q
\end{align}
and the interopolators should be transitive, i.e. for $q < p$:
\begin{align}
\mathbf{S}_{P \leftarrow p} \mathbf{S}_{p \leftarrow q} = \mathbf{S}_{P \leftarrow q}
\end{align}
If these conditions are fulfilled, the updated parallel scheme is shown in Figure \ref{fig:algo_parallel_lr}.
